Sub ProcessAlarmData()
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    
    Dim wsOriginal As Worksheet
    Dim wsSummary As Worksheet
    Dim wsNewSheet1 As Worksheet
    Dim wsFactory As Worksheet
    Dim lastRow As Long
    Dim factoryDict As Object
    Dim factoryAlarmCount As Object
    Dim i As Long
    Dim rowNum As Variant
    Dim factoryRows As Collection
    Dim factoryKey As Variant
    Dim rowsToDelete As Collection
    Dim destRow As Long
    Dim summaryLastRow As Long
    Dim factoryName As String
    Dim key As String
    Dim monitorPoint As String
    Dim indicatorType As String
    Dim sensorModel As String
    Dim dictKeys As Variant
    Dim sheetName As String
    
    On Error GoTo ErrorHandler
    
    ' 使用错误处理避免Sheet1不存在的问题
    On Error Resume Next
    Set wsOriginal = ThisWorkbook.Sheets("Sheet1")
    On Error GoTo ErrorHandler
    
    ' 检查Sheet1是否存在
    If wsOriginal Is Nothing Then
        MsgBox "错误：Sheet1 工作表不存在！", vbExclamation
        GoTo CleanUp
    End If
    
    ' 0. 删除SHELL WOFE 测试工厂的数据
    lastRow = wsOriginal.Cells(wsOriginal.Rows.Count, "A").End(xlUp).Row
    If lastRow > 1 Then
        Set rowsToDelete = New Collection
        
        ' 从下往上遍历，避免删除行导致索引变化
        For i = lastRow To 2 Step -1
            factoryName = wsOriginal.Cells(i, "B").Value ' B列是工厂列
            
            If Not IsError(factoryName) And Not IsEmpty(factoryName) Then
                If Trim(factoryName) = "SHELL WOFE 测试工厂" Then
                    rowsToDelete.Add i
                End If
            End If
        Next i
        
        ' 删除找到的行
        For Each rowNum In rowsToDelete
            wsOriginal.Rows(rowNum).Delete
        Next rowNum
        
        If rowsToDelete.Count > 0 Then
            MsgBox "已删除 " & rowsToDelete.Count & " 行SHELL WOFE测试工厂数据"
        End If
    End If
    
    ' 重新获取最后一行
    lastRow = wsOriginal.Cells(wsOriginal.Rows.Count, "A").End(xlUp).Row
    
    ' 检查是否有数据
    If lastRow < 2 Then
        MsgBox "没有数据需要处理！", vbExclamation
        GoTo CleanUp
    End If
    
    ' 1. 删除N列及之后的列
    If lastRow > 0 Then
        If wsOriginal.Columns.Count >= 14 Then ' 检查N列是否存在
            wsOriginal.Range(wsOriginal.Columns("N"), wsOriginal.Columns(wsOriginal.Columns.Count)).Delete
        End If
    End If
    
    ' 重命名原Sheet1为"全部报警数据"
    wsOriginal.Name = "全部报警数据"
    
    ' 2. 创建工厂报警信息汇总表
    ' 先删除已存在的汇总表（如果存在）
    On Error Resume Next
    Application.DisplayAlerts = False
    ThisWorkbook.Sheets("工厂报警信息汇总").Delete
    Application.DisplayAlerts = True
    On Error GoTo ErrorHandler
    
    Set wsSummary = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    wsSummary.Name = "工厂报警信息汇总"
    
    ' 设置表头
    wsSummary.Range("A1").Value = "工厂"
    wsSummary.Range("B1").Value = "监测点"
    wsSummary.Range("C1").Value = "指标类型"
    wsSummary.Range("D1").Value = "传感器产品型号"
    wsSummary.Range("E1").Value = "报警总数"
    
    ' 使用字典收集唯一数据和报警总数
    Set factoryDict = CreateObject("Scripting.Dictionary")
    Set factoryAlarmCount = CreateObject("Scripting.Dictionary")
    
    ' 首先统计每个工厂的总报警数
    For i = 2 To lastRow
        factoryName = Trim(wsOriginal.Cells(i, "B").Value) ' B列
        
        If Not IsError(factoryName) And Not IsEmpty(factoryName) And factoryName <> "" Then
            If factoryAlarmCount.Exists(factoryName) Then
                factoryAlarmCount(factoryName) = factoryAlarmCount(factoryName) + 1
            Else
                factoryAlarmCount.Add factoryName, 1
            End If
        End If
    Next i
    
    ' 收集唯一数据
    For i = 2 To lastRow
        factoryName = Trim(wsOriginal.Cells(i, "B").Value) ' B列
        
        If Not IsError(factoryName) And Not IsEmpty(factoryName) And factoryName <> "" Then
            monitorPoint = Trim(wsOriginal.Cells(i, "E").Value)
            indicatorType = Trim(wsOriginal.Cells(i, "J").Value)
            sensorModel = Trim(wsOriginal.Cells(i, "H").Value)
            
            If Not IsError(monitorPoint) And Not IsEmpty(monitorPoint) And _
               Not IsError(indicatorType) And Not IsEmpty(indicatorType) And _
               Not IsError(sensorModel) And Not IsEmpty(sensorModel) Then
               
                key = factoryName & "|" & monitorPoint & "|" & indicatorType & "|" & sensorModel
                
                If Not factoryDict.Exists(key) Then
                    factoryDict.Add key, Array(factoryName, monitorPoint, indicatorType, sensorModel, factoryAlarmCount(factoryName))
                End If
            End If
        End If
    Next i
    
    ' 写入汇总数据
    If factoryDict.Count > 0 Then
        dictKeys = factoryDict.Keys
        
        For i = 0 To factoryDict.Count - 1
            With wsSummary
                .Cells(i + 2, 1).Value = factoryDict(dictKeys(i))(0) ' 工厂
                .Cells(i + 2, 2).Value = factoryDict(dictKeys(i))(1) ' 监测点
                .Cells(i + 2, 3).Value = factoryDict(dictKeys(i))(2) ' 指标类型
                .Cells(i + 2, 4).Value = factoryDict(dictKeys(i))(3) ' 传感器产品型号
                .Cells(i + 2, 5).Value = factoryDict(dictKeys(i))(4) ' 报警总数
            End With
        Next i
    End If
    
    ' 3. 创建新的空白Sheet1
    ' 先删除已存在的Sheet1（如果存在）
    On Error Resume Next
    Application.DisplayAlerts = False
    ThisWorkbook.Sheets("Sheet1").Delete
    Application.DisplayAlerts = True
    On Error GoTo ErrorHandler
    
    Set wsNewSheet1 = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    wsNewSheet1.Name = "Sheet1"
    
    ' 设置表头（复制原表的表头）
    wsOriginal.Range("A1:M1").Copy wsNewSheet1.Range("A1")
    
    ' 4. 按工厂拆分数据
    Dim factorySheetsDict As Object
    Set factorySheetsDict = CreateObject("Scripting.Dictionary")
    
    ' 收集所有工厂数据
    For i = 2 To lastRow
        factoryName = Trim(wsOriginal.Cells(i, "B").Value)
        
        If Not IsError(factoryName) And Not IsEmpty(factoryName) And factoryName <> "" Then
            If Not factorySheetsDict.Exists(factoryName) Then
                Set factoryRows = New Collection
                factorySheetsDict.Add factoryName, factoryRows
            Else
                Set factoryRows = factorySheetsDict(factoryName)
            End If
            
            factoryRows.Add i
        End If
    Next i
    
    ' 创建工厂工作表并填充数据
    For Each factoryKey In factorySheetsDict.Keys
        sheetName = Trim(CStr(factoryKey))
        
        ' 确保工作表名称不超过31个字符且不包含非法字符
        sheetName = Left(sheetName, 31)
        sheetName = Replace(sheetName, ":", "_")
        sheetName = Replace(sheetName, "\", "_")
        sheetName = Replace(sheetName, "/", "_")
        sheetName = Replace(sheetName, "?", "_")
        sheetName = Replace(sheetName, "*", "_")
        sheetName = Replace(sheetName, "[", "_")
        sheetName = Replace(sheetName, "]", "_")
        
        ' 删除同名工作表
        On Error Resume Next
        Application.DisplayAlerts = False
        ThisWorkbook.Sheets(sheetName).Delete
        Application.DisplayAlerts = True
        On Error GoTo ErrorHandler
        
        Set wsFactory = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsFactory.Name = sheetName
        
        ' 复制表头
        wsOriginal.Range("A1:M1").Copy wsFactory.Range("A1")
        
        ' 复制数据行
        Set factoryRows = factorySheetsDict(factoryKey)
        destRow = 2
        
        For Each rowNum In factoryRows
            wsOriginal.Range("A" & rowNum & ":M" & rowNum).Copy wsFactory.Range("A" & destRow)
            destRow = destRow + 1
        Next rowNum
        
        ' 排序
        If destRow > 2 Then
            With wsFactory.Sort
                .SortFields.Clear
                .SortFields.Add Key:=wsFactory.Range("J2:J" & destRow - 1), Order:=xlAscending
                .SortFields.Add Key:=wsFactory.Range("M2:M" & destRow - 1), Order:=xlDescending
                .SetRange wsFactory.Range("A1:M" & destRow - 1)
                .Header = xlYes
                .Apply
            End With
        End If
    Next factoryKey
    
    ' 调整列宽
    wsNewSheet1.Columns.AutoFit
    
    ' 为汇总表添加边框和格式
    With wsSummary
        If .Cells(2, 1).Value <> "" Then
            summaryLastRow = .Cells(.Rows.Count, "A").End(xlUp).Row
            
            ' 添加边框
            .Range("A1:E" & summaryLastRow).Borders.LineStyle = xlContinuous
            
            ' 设置标题行格式
            .Range("A1:E1").Interior.Color = RGB(200, 200, 200)
            .Range("A1:E1").Font.Bold = True
            
            ' 自动调整列宽
            .Columns("A:E").AutoFit
        End If
    End With
    
    ' 5. 移动新的Sheet1到第三个位置（在汇总表之后）
    If ThisWorkbook.Sheets.Count >= 3 Then
        On Error Resume Next
        wsNewSheet1.Move After:=ThisWorkbook.Sheets("工厂报警信息汇总")
        On Error GoTo ErrorHandler
    End If
    
CleanUp:
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    
    MsgBox "数据处理完成！" & vbCrLf & _
           "已删除SHELL WOFE测试工厂数据" & vbCrLf & _
           "原Sheet1已重命名为'全部报警数据'" & vbCrLf & _
           "已创建空白Sheet1工作表（含表头）" & vbCrLf & _
           "已按工厂拆分数据到各自的工作表", vbInformation
    Exit Sub
    
ErrorHandler:
    MsgBox "发生错误: " & Err.Description & vbCrLf & _
           "错误号: " & Err.Number, vbCritical
    Resume CleanUp
End Sub
